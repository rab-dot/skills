## 1. Project bootstrap

```bash
npm create vite@latest my-app -- --template react-ts
cd my-app
npm install
```

### 1.1 Install Tailwind v4 + Vite plugin

```bash
npm install tailwindcss @tailwindcss/vite
npm install -D @types/node
```

### 1.2 Configure `vite.config.ts`

```ts
import path from 'path'
import react from '@vitejs/plugin-react'
import tailwindcss from '@tailwindcss/vite'
import { defineConfig } from 'vite'

export default defineConfig({
  plugins: [react(), tailwindcss()],
  resolve: {
    alias: { '@': path.resolve(__dirname, './src') },
  },
})
```

### 1.3 Configure `tsconfig.app.json`

```json
{
  "compilerOptions": {
    "baseUrl": ".",
    "paths": { "@/*": ["./src/*"] }
  }
}
```

### 1.4 `src/index.css` — Tailwind v4 import (NO directives)

```css
@import "tailwindcss";
```

> **TRAP**: Tailwind v4 uses `@import "tailwindcss"` NOT the old
> `@tailwind base/components/utilities` directives. Using old directives BREAKS v4.

### 1.5 Initialize shadcn/ui

```bash
npx shadcn@latest init
```

CLI prompts: choose style (Default/New York), base color, CSS variables — yes.

## 3. `cn()` utility — always use for conditional classes

```ts
// src/lib/utils.ts  (auto-generated by shadcn init)
import { clsx, type ClassValue } from 'clsx'
import { twMerge } from 'tailwind-merge'

export function cn(...inputs: ClassValue[]) {
  return twMerge(clsx(inputs))
}
```

```tsx
// Usage
<div className={cn('base-class', isActive && 'active-class', className)} />
```

Never concatenate Tailwind classes with template literals — always use `cn()`.

## 5. Server state — TanStack Query v5

```bash
npm install @tanstack/react-query @tanstack/react-query-devtools
```

```ts
// src/lib/queryClient.ts
import { QueryClient } from '@tanstack/react-query'

export const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      staleTime: 1000 * 60 * 5,   // 5 min
      gcTime:    1000 * 60 * 30,  // 30 min (renamed from cacheTime in v5)
      retry: 1,
      refetchOnWindowFocus: false,
    },
  },
})
```

```tsx
// src/main.tsx — wrap app
import { QueryClientProvider } from '@tanstack/react-query'
import { ReactQueryDevtools } from '@tanstack/react-query-devtools'

<QueryClientProvider client={queryClient}>
  <RouterProvider router={router} />
  <ReactQueryDevtools initialIsOpen={false} />
</QueryClientProvider>
```

```ts
// src/features/automations/automationQueries.ts
import { queryOptions } from '@tanstack/react-query'
import { api } from '@/lib/api'

export const automationsQueryOptions = () =>
  queryOptions({
    queryKey: ['automations'],
    queryFn: () => api.get('/automations').then(r => r.data),
  })
```

```tsx
// In component — v5 hook
import { useSuspenseQuery } from '@tanstack/react-query'

const { data } = useSuspenseQuery(automationsQueryOptions())
```

> **v5 TRAPS**: `cacheTime` renamed to `gcTime`. `useQuery` returns `{ data, isPending }` — `isLoading` deprecated. Use `useSuspenseQuery` for Suspense boundaries.

## 6. Client state — Zustand v5

```bash
npm install zustand
```

```ts
// src/store/useUIStore.ts
import { create } from 'zustand'

interface UIState {
  isSidebarOpen: boolean
  toggleSidebar: () => void
  activeAccountId: string | null
  setActiveAccount: (id: string) => void
}

export const useUIStore = create<UIState>((set) => ({
  isSidebarOpen: true,
  toggleSidebar: () => set(s => ({ isSidebarOpen: !s.isSidebarOpen })),
  activeAccountId: null,
  setActiveAccount: (id) => set({ activeAccountId: id }),
}))
```

**Rule**: Server data (API responses) → TanStack Query. UI state (sidebar, modals, selected rows) → Zustand.

## 7. Forms — React Hook Form v7 + Zod v3

```bash
npm install react-hook-form zod @hookform/resolvers
```

```tsx
import { useForm } from 'react-hook-form'
import { zodResolver } from '@hookform/resolvers/zod'
import { z } from 'zod'
import {
  Form, FormControl, FormField, FormItem, FormLabel, FormMessage,
} from '@/components/ui/form'
import { Input } from '@/components/ui/input'
import { Button } from '@/components/ui/button'

const schema = z.object({
  keyword: z.string().min(1, 'Required'),
  replyText: z.string().min(5, 'Minimum 5 characters'),
})

type FormValues = z.infer<typeof schema>

export function AutomationForm() {
  const form = useForm<FormValues>({
    resolver: zodResolver(schema),
    defaultValues: { keyword: '', replyText: '' },
  })

  const onSubmit = (data: FormValues) => { /* call API */ }

  return (
    <Form {...form}>
      <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
        <FormField control={form.control} name="keyword" render={({ field }) => (
          <FormItem>
            <FormLabel>Keyword</FormLabel>
            <FormControl><Input {...field} /></FormControl>
            <FormMessage />
          </FormItem>
        )} />
        <Button type="submit">Save</Button>
      </form>
    </Form>
  )
}
```

## 8. HTTP client — Axios v1

```bash
npm install axios
```

```ts
// src/lib/api.ts
import axios from 'axios'

export const api = axios.create({
  baseURL: import.meta.env.VITE_API_URL ?? 'http://localhost:3001/api',
  headers: { 'Content-Type': 'application/json' },
})

// Auth interceptor
api.interceptors.request.use(cfg => {
  const token = localStorage.getItem('token')
  if (token) cfg.headers.Authorization = `Bearer ${token}`
  return cfg
})

// Error interceptor
api.interceptors.response.use(
  r => r,
  err => {
    if (err.response?.status === 401) {
      window.location.href = '/login'
    }
    return Promise.reject(err)
  },
)
```

```tsx
import { ReactFlow, addEdge, useNodesState, useEdgesState } from '@xyflow/react'

const initialNodes = [
  { id: '1', position: { x: 0, y: 0 }, data: { label: 'Trigger: Comment' } },
  { id: '2', position: { x: 0, y: 150 }, data: { label: 'Action: Send DM' } },
]

export function AutomationBuilder() {
  const [nodes, , onNodesChange] = useNodesState(initialNodes)
  const [edges, setEdges, onEdgesChange] = useEdgesState([])

  return (
    <div style={{ height: 600 }}>
      <ReactFlow
        nodes={nodes}
        edges={edges}
        onNodesChange={onNodesChange}
        onEdgesChange={onEdgesChange}
        onConnect={params => setEdges(es => addEdge(params, es))}
        fitView
      />
    </div>
  )
}
```

## Version traps

- **Tailwind v4**: `@import "tailwindcss"` in CSS — NO old `@tailwind` directives.
- **Tailwind v4**: Config is in CSS via `@theme {}` block — NO `tailwind.config.js`.
- **TanStack Query v5**: `gcTime` (not `cacheTime`); `isPending` (not `isLoading` for fresh queries).
- **Zustand v5**: `create<T>()` without `devtools` wrapper by default — add explicitly if needed.
- **React Router v7**: Package is `react-router` (unified) — NOT `react-router-dom` separately.
- **@xyflow/react v12**: Package renamed from `reactflow` to `@xyflow/react`.
- **shadcn/ui components**: Stored in `src/components/ui/` — edit freely (you own the code).
- **lucide-react**: Import icons individually, never `import * from 'lucide-react'`.

## Checklist

- [ ] Server state in TanStack Query, UI state in Zustand — never mixed.

- [ ] Tailwind v4 `@import "tailwindcss"` in `index.css`.
- [ ] Path alias `@/` configured in both `vite.config.ts` and `tsconfig.app.json`.
